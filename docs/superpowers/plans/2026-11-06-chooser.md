# Chooser (« Quel outil pour moi ? ») Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a static, three-question chooser that returns one recommended Yoto tool plus two alternatives, with readable reasons and a shareable URL.

**Architecture:** Pure scoring lives in `site/src/lib/chooser.ts` and is unit-tested in Node. The page is statically built by Astro; because `Astro.url.search` is empty at build time on a static host, results are rendered client-side from the URL and the no-JS fallback is a server-rendered list of card-making tools plus goal filter links. The client script is a progressive enhancement over a real `GET` form.

**Tech Stack:** Astro 7 (static output), TypeScript, Vitest, Playwright, pnpm 10.

**Spec:** `docs/roadmap.md` §4.1 (T1–T6) and §3 (vague 1).

## Global Constraints

- 100 % static, GitHub Pages: no backend, no account, no analytics, no runtime network request from the browser.
- No migration of the 58 project YAML files and no schema change. Derive everything from `category`, `tags`, `platforms`, `requires`, `warning`, `status`, `official`, `meta.stars`.
- Never recommend a project whose status is `archived` or `deprecated`.
- Every recommendation shows at least one readable reason; never display a raw score.
- EN/FR route parity is mandatory (`pnpm check:i18n`).
- Every new i18n key must exist in both `site/src/i18n/en.ts` and `site/src/i18n/fr.ts` (enforced by `site/src/i18n/index.test.ts`).
- The chooser client script must stay under **6 KB gzip**; the site total must stay under **30 KB gzip** (`pnpm check:budget`).
- Node `>=22.23.4`, pnpm `10.11.0`.
- Commit after every task. Use Conventional Commits (`feat:`, `test:`, `docs:`).

## Decisions taken before implementation

Four points in the roadmap are resolved here, because the code cannot be written unambiguously otherwise:

1. **Static hosting vs. server-rendered results.** Astro outputs static HTML; a build cannot read `?goal=…`. The roadmap's T2 sentence “les résultats si l'URL contient déjà un état” is therefore implemented by the client script on load, not by Astro. The no-JS fallback is the static list of card-making tools + goal filter links, which is exactly what the acceptance criteria require (“Sans JS : la liste statique et les liens de filtres restent utilisables”).
2. **Platform semantics.** `web` and `cli` count as usable on Windows/macOS/Linux. `ios` and `android` only match their exact platform. `self-hosted` matches `requires: [self-hosted]`. `any` gives no platform bonus at all, so unrelated projects do not enter the ranking.
3. **No `bestFor` field.** A simulation against the real 58 projects showed the existing fields already produce correct results for the six personas (see Task 5). The roadmap's fallback field stays unbuilt; it is not needed for v1.
4. **Platform bonus requires goal relevance.** `worksOn` alone no longer earns
   points: platform fit is a tie-breaker among projects that already match the
   goal's category or tags. Without this, selecting `goal=backup&platform=macos`
   could pad the list with unrelated web tools.

```ts
import type { Platform, Project } from '../../../scripts/lib/types';
import type { Lang } from './routes';

export type Goal =
  'youtube' | 'files' | 'podcast' | 'audiobook' | 'backup' | 'automate' | 'develop';
export type Level = 'simple' | 'advanced';
export type ChooserPlatform = Platform | 'self-hosted' | 'any';

export interface ChooserState {
  goal: Goal;
  platform: ChooserPlatform;
  level: Level;
}

/**
 * The subset of a project the chooser needs. The full Project is assignable,
 * and the same shape can be serialised into the page for the client script.
 */
export interface ChooserProject extends Pick<
  Project,
  | 'slug'
  | 'name'
  | 'description'
  | 'category'
  | 'tags'
  | 'platforms'
  | 'requires'
  | 'warning'
  | 'status'
  | 'official'
  | 'meta'
> {
  descriptionFr?: string;
  statusLabel?: string;
}

export type ReasonCode = 'goal' | 'category' | 'platform' | 'simple' | 'official';

export interface Recommendation {
  project: ChooserProject;
  score: number;
  reasons: ReasonCode[];
}

const GOAL_RULES: Record<Goal, { categories: string[]; tags: string[] }> = {
  youtube: { categories: ['create'], tags: ['youtube', 'myo'] },
  files: { categories: ['create'], tags: ['myo', 'audiobook'] },
  podcast: { categories: ['audio'], tags: ['podcast', 'rss'] },
  audiobook: { categories: ['audio', 'create'], tags: ['audiobook'] },
  backup: { categories: ['backup'], tags: [] },
  automate: { categories: ['automate'], tags: [] },
  develop: { categories: ['develop'], tags: [] },
};

const ADVANCED_REQUIRES = new Set(['yoto-api-key', 'self-hosted']);
const DESKTOP_PLATFORMS = new Set(['windows', 'macos', 'linux']);

function worksOn(project: ChooserProject, platform: ChooserPlatform): boolean {
  if (platform === 'any') return true;
  if (platform === 'self-hosted') return project.requires?.includes('self-hosted') ?? false;
  if (project.platforms.includes(platform)) return true;
  if (DESKTOP_PLATFORMS.has(platform)) {
    return project.platforms.includes('web') || project.platforms.includes('cli');
  }
  return false;
}

function isAdvanced(project: ChooserProject): boolean {
  return (
    (project.requires?.some((requirement) => ADVANCED_REQUIRES.has(requirement)) ?? false) ||
    project.platforms.includes('cli') ||
    project.warning === 'experimental'
  );
}

export function scoreProject(project: ChooserProject, state: ChooserState): number {
  const rule = GOAL_RULES[state.goal];
  const goalRelevant =
    rule.categories.includes(project.category) ||
    project.tags.some((tag) => rule.tags.includes(tag));
  let score = 0;
  if (rule.categories.includes(project.category)) score += 4;
  score += project.tags.filter((tag) => rule.tags.includes(tag)).length * 3;
  if (goalRelevant && state.platform !== 'any' && worksOn(project, state.platform)) score += 3;
  if (project.status === 'stale') score -= 5;
  if (state.level === 'simple' && isAdvanced(project)) score -= 3;
  return score;
}

function reasonsFor(project: ChooserProject, state: ChooserState): ReasonCode[] {
  const rule = GOAL_RULES[state.goal];
  const reasons: ReasonCode[] = [];
  if (project.tags.some((tag) => rule.tags.includes(tag))) reasons.push('goal');
  if (rule.categories.includes(project.category)) reasons.push('category');
  if (state.platform !== 'any' && worksOn(project, state.platform)) reasons.push('platform');
  if (state.level === 'simple' && !isAdvanced(project)) reasons.push('simple');
  if (project.official) reasons.push('official');
  return reasons.slice(0, 3);
}

function compare(
  a: { project: ChooserProject; score: number },
  b: { project: ChooserProject; score: number },
): number {
  return (
    b.score - a.score ||
    (b.project.meta?.stars ?? 0) - (a.project.meta?.stars ?? 0) ||
    a.project.name.localeCompare(b.project.name, 'en')
  );
}

export function recommendProjects(
  projects: ChooserProject[],
  state: ChooserState,
  limit = 3,
): Recommendation[] {
  return projects
    .filter((project) => project.status !== 'archived' && project.status !== 'deprecated')
    .map((project) => ({ project, score: scoreProject(project, state) }))
    .filter((entry) => entry.score > 0)
    .sort(compare)
    .slice(0, limit)
    .map((entry) => ({ ...entry, reasons: reasonsFor(entry.project, state) }));
}
```

```ts
export interface ChooserOption<T extends string> {
  id: T;
  label: Record<Lang, string>;
}

export const GOALS: ChooserOption<Goal>[] = [
  { id: 'youtube', label: { en: 'Turn YouTube into a card', fr: 'Transformer YouTube en carte' } },
  {
    id: 'files',
    label: { en: 'Turn audio files into a card', fr: 'Transformer des fichiers audio en carte' },
  },
  {
    id: 'podcast',
    label: { en: 'Turn a podcast into a card', fr: 'Transformer un podcast en carte' },
  },
  {
    id: 'audiobook',
    label: { en: 'Turn an audiobook into a card', fr: 'Transformer un livre audio en carte' },
  },
  { id: 'backup', label: { en: 'Back up my cards', fr: 'Sauvegarder mes cartes' } },
  { id: 'automate', label: { en: 'Automate my player', fr: 'Automatiser mon lecteur' } },
  { id: 'develop', label: { en: 'Build with the API', fr: 'Développer avec l’API' } },
];

export const PLATFORMS: ChooserOption<ChooserPlatform>[] = [
  { id: 'any', label: { en: 'No preference', fr: 'Peu importe' } },
  { id: 'web', label: { en: 'Web browser', fr: 'Navigateur web' } },
  { id: 'windows', label: { en: 'Windows', fr: 'Windows' } },
  { id: 'macos', label: { en: 'macOS', fr: 'macOS' } },
  { id: 'linux', label: { en: 'Linux', fr: 'Linux' } },
  { id: 'android', label: { en: 'Android', fr: 'Android' } },
  { id: 'ios', label: { en: 'iOS', fr: 'iOS' } },
  { id: 'self-hosted', label: { en: 'Self-hosted server', fr: 'Serveur auto-hébergé' } },
];

export const LEVELS: ChooserOption<Level>[] = [
  {
    id: 'simple',
    label: {
      en: 'I don’t want to install anything',
      fr: 'Je ne veux rien installer',
    },
  },
  {
    id: 'advanced',
    label: {
      en: 'I’m comfortable with a terminal',
      fr: 'Je suis à l’aise avec un terminal',
    },
  },
];

export function parseChooserState(search: string): ChooserState | null {
  const params = new URLSearchParams(search.startsWith('?') ? search.slice(1) : search);
  const goal = GOALS.find((option) => option.id === params.get('goal'))?.id;
  const platform = PLATFORMS.find((option) => option.id === params.get('platform'))?.id;
  const level = LEVELS.find((option) => option.id === params.get('level'))?.id;
  return goal && platform && level ? { goal, platform, level } : null;
}

export function serializeChooserState(state: ChooserState): string {
  return new URLSearchParams({
    goal: state.goal,
    platform: state.platform,
    level: state.level,
  }).toString();
}

const GOAL_NOUNS: Record<Goal, Record<Lang, string>> = {
  youtube: { en: 'YouTube', fr: 'YouTube' },
  files: { en: 'audio files', fr: 'les fichiers audio' },
  podcast: { en: 'podcasts', fr: 'les podcasts' },
  audiobook: { en: 'audiobooks', fr: 'les livres audio' },
  backup: { en: 'backups', fr: 'la sauvegarde' },
  automate: { en: 'automation', fr: 'l’automatisation' },
  develop: { en: 'development', fr: 'le développement' },
};

const PLATFORM_NOUNS: Record<Exclude<ChooserPlatform, 'any'>, Record<Lang, string>> = {
  web: { en: 'the browser', fr: 'le navigateur' },
  windows: { en: 'Windows', fr: 'Windows' },
  macos: { en: 'macOS', fr: 'macOS' },
  linux: { en: 'Linux', fr: 'Linux' },
  android: { en: 'Android', fr: 'Android' },
  ios: { en: 'iOS', fr: 'iOS' },
  cli: { en: 'the terminal', fr: 'le terminal' },
  'home-assistant': { en: 'Home Assistant', fr: 'Home Assistant' },
  hardware: { en: 'dedicated hardware', fr: 'un matériel dédié' },
  api: { en: 'the API', fr: 'l’API' },
  'self-hosted': { en: 'your own server', fr: 'votre serveur' },
};

export function reasonLabels(rec: Recommendation, state: ChooserState, lang: Lang): string[] {
  return rec.reasons.map((reason) => {
    switch (reason) {
      case 'goal':
        return lang === 'fr'
          ? `Pensé pour ${GOAL_NOUNS[state.goal].fr}`
          : `Made for ${GOAL_NOUNS[state.goal].en}`;
      case 'category':
        return lang === 'fr' ? 'Adapté à cette tâche' : 'Fits this task';
      case 'platform': {
        const platform = state.platform;
        if (platform === 'any') return lang === 'fr' ? 'Compatible' : 'Compatible';
        if (platform === 'web') {
          return lang === 'fr' ? 'Fonctionne dans le navigateur' : 'Runs in the browser';
        }
        const noun = PLATFORM_NOUNS[platform][lang];
        return lang === 'fr' ? `Fonctionne sur ${noun}` : `Works on ${noun}`;
      }
      case 'simple':
        return lang === 'fr' ? 'Sans terminal ni clé API' : 'No terminal or API key';
      case 'official':
        return lang === 'fr' ? 'Outil officiel Yoto' : 'Official Yoto tool';
    }
  });
}

export function goalFilterHref(goal: Goal, base = '/projects/'): string {
  const rule = GOAL_RULES[goal];
  const params = new URLSearchParams();
  if (rule.categories[0]) params.set('cat', rule.categories[0]);
  if (rule.tags[0]) params.set('tag', rule.tags[0]);
  return `${base}?${params.toString()}`;
}

export function projectsFilterHref(state: ChooserState, base = '/projects/'): string {
  return goalFilterHref(state.goal, base);
}

const MYO_STEPS: Partial<Record<Goal, number>> = {
  youtube: 1,
  files: 2,
  podcast: 1,
  audiobook: 1,
};

export function myoStep(goal: Goal): number | undefined {
  return MYO_STEPS[goal];
}
```

- [ ] **Step 3: Implement the script**

Create `site/src/scripts/chooser.ts`:

```ts
import {
  myoStep,
  parseChooserState,
  projectsFilterHref,
  reasonLabels,
  recommendProjects,
  serializeChooserState,
  type ChooserProject,
  type ChooserState,
} from '../lib/chooser';
import { localePath } from '../lib/routes';
import type { Lang } from '../lib/routes';

const dataElement = document.getElementById('chooser-data');
const form = document.querySelector<HTMLFormElement>('[data-chooser]');
const region = document.querySelector<HTMLElement>('[data-results-region]');
const list = document.querySelector<HTMLOListElement>('[data-results]');
const empty = document.querySelector<HTMLElement>('[data-results-empty]');
const title = document.querySelector<HTMLElement>('[data-results-title]');
const template = document.querySelector<HTMLTemplateElement>('[data-result-template]');

if (dataElement && form && region && list && empty && title && template) {
  const projects = JSON.parse(dataElement.textContent ?? '[]') as ChooserProject[];
  const lang: Lang = document.documentElement.lang === 'fr' ? 'fr' : 'en';
  const projectsBase = dataElement.dataset.projectsBase ?? localePath(lang, '/projects/');
  const myoBase = dataElement.dataset.myoBase ?? localePath(lang, '/myo/');
  let complete = parseChooserState(window.location.search) !== null;

  const readState = (): ChooserState | null => {
    const values = new FormData(form);
    const params = new URLSearchParams();
    for (const key of ['goal', 'platform', 'level']) {
      const value = values.get(key);
      if (typeof value === 'string' && value) params.set(key, value);
    }
    return parseChooserState(params.toString());
  };

  const syncForm = (state: ChooserState) => {
    for (const [name, value] of Object.entries(state)) {
      const input = form.querySelector<HTMLInputElement>(`input[name="${name}"][value="${value}"]`);
      if (input) input.checked = true;
    }
  };

  const render = (state: ChooserState | null, moveFocus = false) => {
    if (!state) {
      region.hidden = true;
      return;
    }

    const recommendations = recommendProjects(projects, state);
    const items = recommendations.flatMap((rec, index) => {
      const item = template.content.firstElementChild?.cloneNode(true) as HTMLElement | undefined;
      if (!item) return [];

      const link = item.querySelector<HTMLAnchorElement>('[data-link]');
      if (link) {
        link.href = localePath(lang, `/p/${rec.project.slug}/`);
        link.textContent = rec.project.name;
      }

      const rank = item.querySelector<HTMLElement>('[data-rank]');
      if (rank) {
        rank.textContent =
          index === 0 ? (template.dataset.labelBest ?? '') : (template.dataset.labelAlt ?? '');
      }

      const description = item.querySelector<HTMLElement>('[data-description]');
      if (description) description.textContent = rec.project.description;

      const status = item.querySelector<HTMLElement>('[data-status]');
      if (status) {
        status.textContent = rec.project.statusLabel ?? '';
        status.hidden = rec.project.status === 'active';
      }

      const reasons = item.querySelector<HTMLElement>('[data-reasons]');
      if (reasons) {
        reasons.replaceChildren(
          ...reasonLabels(rec, state, lang).map((reason) => {
            const entry = document.createElement('li');
            entry.textContent = reason;
            return entry;
          }),
        );
      }

      const projectLink = item.querySelector<HTMLAnchorElement>('[data-link-project]');
      if (projectLink) {
        projectLink.href = localePath(lang, `/p/${rec.project.slug}/`);
        projectLink.textContent = template.dataset.labelProject ?? '';
      }

      const myoLink = item.querySelector<HTMLAnchorElement>('[data-link-myo]');
      const step = myoStep(state.goal);
      if (myoLink) {
        myoLink.hidden = step === undefined;
        if (step !== undefined) {
          myoLink.href = `${myoBase}#step-${step}`;
          myoLink.textContent = template.dataset.labelMyo ?? '';
        }
      }

      const filtersLink = item.querySelector<HTMLAnchorElement>('[data-link-filters]');
      if (filtersLink) {
        filtersLink.href = projectsFilterHref(state, projectsBase);
        filtersLink.textContent = template.dataset.labelFilters ?? '';
      }

      return [item];
    });

    list.replaceChildren(...items);
    list.hidden = items.length === 0;
    empty.hidden = items.length > 0;
    region.hidden = false;

    const query = serializeChooserState(state);
    window.history.replaceState(
      null,
      '',
      `${window.location.pathname}?${query}${window.location.hash}`,
    );

    if (moveFocus) title.focus();
  };

  const initial = parseChooserState(window.location.search);
  if (initial) syncForm(initial);
  render(initial);

  form.addEventListener('change', () => {
    const state = readState();
    const becameComplete = !complete && state !== null;
    complete = state !== null;
    render(state, becameComplete);
  });

  form.addEventListener('submit', (event) => {
    event.preventDefault();
    const state = readState();
    complete = state !== null;
    render(state, true);
  });
}
```

```ts
import { describe, expect, it } from 'vitest';
import { createLoader } from '../scripts/lib/load';
import {
  recommendProjects,
  type ChooserPlatform,
  type ChooserState,
  type Goal,
  type Level,
} from '../site/src/lib/chooser';

const projects = createLoader().projects();
const slugsFor = (state: ChooserState) =>
  recommendProjects(projects, state).map((rec) => rec.project.slug);
const categoryOf = (slug: string) => projects.find((project) => project.slug === slug)?.category;

describe('chooser personas', () => {
  it('parent: YouTube on macOS, no install', () => {
    const slugs = slugsFor({ goal: 'youtube', platform: 'macos', level: 'simple' });
    expect(slugs[0]).toBe('louis');
    for (const slug of slugs) expect(categoryOf(slug), slug).toBe('create');
  });

  it('Windows user: YouTube, no install', () => {
    const slugs = slugsFor({ goal: 'youtube', platform: 'windows', level: 'simple' });
    expect(slugs[0]).toBe('louis');
    expect(slugs).toContain('yoto-media-manager');
  });

  it('Mac user: audiobook, no install', () => {
    expect(slugsFor({ goal: 'audiobook', platform: 'macos', level: 'simple' })[0]).toBe(
      'yoto-book-maker',
    );
  });

  it('self-hoster: podcast', () => {
    const slugs = slugsFor({ goal: 'podcast', platform: 'self-hosted', level: 'advanced' });
    expect(slugs[0]).toBe('podcast-catalogue');
    for (const slug of slugs) expect(categoryOf(slug), slug).toBe('audio');
  });

  it('developer: API tools', () => {
    const slugs = slugsFor({ goal: 'develop', platform: 'any', level: 'advanced' });
    expect(slugs[0]).toBe('yoto-api');
    for (const slug of slugs) expect(categoryOf(slug), slug).toBe('develop');
  });

  it('backup: only backup tools, never a stale one first', () => {
    const slugs = slugsFor({ goal: 'backup', platform: 'any', level: 'simple' });
    expect(slugs[0]).toBe('yoto-tools');
    for (const slug of slugs) expect(categoryOf(slug), slug).toBe('backup');
  });

  it('automation: never recommends the deprecated integration', () => {
    const slugs = slugsFor({ goal: 'automate', platform: 'any', level: 'simple' });
    for (const slug of slugs) expect(categoryOf(slug), slug).toBe('automate');
    expect(slugs).not.toContain('yoto-ha');
  });

  it('never returns archived or deprecated projects for any answer combination', () => {
    const goals: Goal[] = [
      'youtube',
      'files',
      'podcast',
      'audiobook',
      'backup',
      'automate',
      'develop',
    ];
    const platforms: ChooserPlatform[] = [
      'any',
      'web',
      'windows',
      'macos',
      'linux',
      'android',
      'ios',
      'self-hosted',
    ];
    const levels: Level[] = ['simple', 'advanced'];

    for (const goal of goals) {
      for (const platform of platforms) {
        for (const level of levels) {
          const recs = recommendProjects(projects, { goal, platform, level });
          const slugs = recs.map((rec) => rec.project.slug);
          expect(recs.length).toBeLessThanOrEqual(3);
          expect(new Set(slugs).size).toBe(slugs.length);
          for (const rec of recs) {
            expect(['archived', 'deprecated']).not.toContain(rec.project.status);
            expect(rec.reasons.length).toBeGreaterThan(0);
            expect(rec.score).toBeGreaterThan(0);
          }
        }
      }
    }
  });

  it('computes a recommendation in well under 50 ms', () => {
    const start = performance.now();
    for (let index = 0; index < 100; index += 1) {
      recommendProjects(projects, { goal: 'youtube', platform: 'macos', level: 'simple' });
    }
    expect(performance.now() - start).toBeLessThan(50);
  });
});
```
